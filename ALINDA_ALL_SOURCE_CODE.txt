  <script type="module">
    import { auth, db } from "./firebase-config.js";
    import { signInWithEmailAndPassword } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-auth.js";
    import {
      collection, getDocs, doc, getDoc, addDoc, serverTimestamp
    } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-firestore.js";

    const byId = (id) => document.getElementById(id);
    const showModal = (id) => byId(id)?.classList.add("open");
    const hideModal = (id) => byId(id)?.classList.remove("open");
    const money = (value) => `UGX ${Number(value || 0).toLocaleString("en-UG")}`;

    function addCell(row, value, strong = false) {
      const cell = document.createElement("td");
      if (strong) {
        const b = document.createElement("strong");
        b.textContent = value ?? "—";
        cell.appendChild(b);
      } else {
        cell.textContent = value ?? "—";
      }
      row.appendChild(cell);
    }

    async function loadPublicSchoolData() {
      const noticesContainer = byId("liveSchoolNotices");
      const unebBody = byId("unebTableBody");
      const feesBody = byId("feesTableBody");

      try {
        const [noticesSnap, unebSnap, feesSnap] = await Promise.all([
          getDocs(collection(db, "school_notices")),
          getDocs(collection(db, "uneb_results")),
          getDocs(collection(db, "fees_structure"))
        ]);

        noticesContainer.replaceChildren();
        if (noticesSnap.empty) {
          const card = document.createElement("div");
          card.className = "card";
          const title = document.createElement("h4");
          title.textContent = "School updates";
          const message = document.createElement("p");
          message.textContent = "New notices from the school administration will appear here.";
          card.append(title, message);
          noticesContainer.appendChild(card);
        } else {
          noticesSnap.forEach((itemDoc) => {
            const item = itemDoc.data();
            const card = document.createElement("article");
            card.className = "card gold-top";
            const author = document.createElement("small");
            author.className = "eyebrow";
            author.textContent = item.postedBy || "Administration";
            const title = document.createElement("h4");
            title.textContent = item.title || "School notice";
            const body = document.createElement("p");
            body.textContent = item.body || "";
            card.append(author, title, body);
            noticesContainer.appendChild(card);
          });
        }

        unebBody.replaceChildren();
        if (unebSnap.empty) {
          unebBody.innerHTML = '<tr><td colspan="6">PLE results will appear here when officially published by the school.</td></tr>';
        } else {
          unebSnap.forEach((itemDoc) => {
            const r = itemDoc.data();
            const row = document.createElement("tr");
            addCell(row, r.year, true);
            addCell(row, r.candidates);
            addCell(row, r.div1);
            addCell(row, r.div2);
            addCell(row, r.passRate == null ? "—" : `${r.passRate}%`, true);
            addCell(row, r.bestAgg);
            unebBody.appendChild(row);
          });
        }

        feesBody.replaceChildren();
        if (feesSnap.empty) {
          feesBody.innerHTML = '<tr><td colspan="5">The school office will publish the current termly fees structure here.</td></tr>';
        } else {
          feesSnap.forEach((itemDoc) => {
            const f = itemDoc.data();
            const row = document.createElement("tr");
            addCell(row, f.section, true);
            addCell(row, money(f.dayFee));
            addCell(row, money(f.boardingFee));
            addCell(row, money(f.foodFee));
            addCell(row, money(f.uniformFee));
            feesBody.appendChild(row);
          });
        }
      } catch (error) {
        console.error("Unable to load public school information:", error);
        noticesContainer.textContent = "School updates are temporarily unavailable. Please try again later.";
        unebBody.innerHTML = '<tr><td colspan="6">Results are temporarily unavailable.</td></tr>';
        feesBody.innerHTML = '<tr><td colspan="5">Fees information is temporarily unavailable.</td></tr>';
      }
    }

    loadPublicSchoolData();

    byId("openVerifyBtn")?.addEventListener("click", () => showModal("verifyModal"));
    byId("unlockExtraBtn")?.addEventListener("click", () => showModal("verifyModal"));
    byId("closeVerifyModal")?.addEventListener("click", () => hideModal("verifyModal"));
    byId("openAdmissionModalBtn")?.addEventListener("click", () => showModal("admissionModal"));
    byId("closeAdmissionModal")?.addEventListener("click", () => hideModal("admissionModal"));
    byId("openPortalLoginBtn")?.addEventListener("click", () => showModal("loginModal"));
    byId("closeLoginModal")?.addEventListener("click", () => hideModal("loginModal"));

    ["verifyModal", "admissionModal", "loginModal"].forEach((id) => {
      byId(id)?.addEventListener("click", (event) => {
        if (event.target.id === id) hideModal(id);
      });
    });

    if (sessionStorage.getItem("alinda_verified") === "true") {
      byId("unlockedContent").style.display = "block";
      byId("verifyStatusText").textContent = "Verified visitor session active.";
    }

    byId("visitorVerifyForm").addEventListener("submit", async (event) => {
      event.preventDefault();
      const message = byId("verifyMsg");
      message.textContent = "Checking your access code…";
      try {
        // A passcode must be configured in Firestore settings/publicAccess.
        // No hard-coded fallback code is used.
        const configDoc = await getDoc(doc(db, "settings", "publicAccess"));
        const validCode = configDoc.exists() ? configDoc.data().visitorPasscode : null;
        const enteredCode = byId("visitorCode").value.trim();

        if (!validCode || enteredCode !== validCode) {
          message.textContent = "The code could not be verified. Please contact the school office.";
          return;
        }

        await addDoc(collection(db, "verified_visitors"), {
          name: byId("visitorName").value.trim(),
          contact: byId("visitorContact").value.trim(),
          verifiedAt: serverTimestamp()
        });
        sessionStorage.setItem("alinda_verified", "true");
        byId("unlockedContent").style.display = "block";
        byId("verifyStatusText").textContent = "Visitor access verified for this browser session.";
        byId("visitorVerifyForm").reset();
        message.textContent = "";
        hideModal("verifyModal");
      } catch (error) {
        console.error("Visitor verification failed:", error);
        message.textContent = "Verification is temporarily unavailable. Please contact the school office.";
      }
    });

    byId("publicAdmissionForm").addEventListener("submit", async (event) => {
      event.preventDefault();
      const message = byId("admMsg");
      message.textContent = "Submitting application…";
      try {
        await addDoc(collection(db, "admissions"), {
          childName: byId("admChildName").value.trim(),
          childClass: byId("admChildClass").value,
          parentName: byId("admParentName").value.trim(),
          parentPhone: byId("admParentPhone").value.trim(),
          status: "Pending",
          submittedAt: serverTimestamp()
        });
        message.textContent = "Application received. The school office will contact you.";
        byId("publicAdmissionForm").reset();
      } catch (error) {
        console.error("Admission submission failed:", error);
        message.textContent = "We could not submit the application. Please try again later.";
      }
    });

    byId("firebaseLoginForm").addEventListener("submit", async (event) => {
      event.preventDefault();
      const message = byId("loginMsg");
      message.textContent = "Signing in…";
      try {
        const credential = await signInWithEmailAndPassword(
          auth, byId("loginEmail").value.trim(), byId("loginPassword").value
        );
        const userDoc = await getDoc(doc(db, "users", credential.user.uid));
        if (!userDoc.exists()) {
          message.textContent = "This account has no school role. Please contact the administrator.";
          return;
        }
        const routes = {
          pupil: "pupil.html",
          staff: "staff.html",
          dos: "dos.html",
          bursar: "bursar.html",
          headteacher: "headteacher.html",
          director: "directors.html"
        };
        const destination = routes[userDoc.data().role];
        if (!destination) {
          message.textContent = "This account has an unrecognized role. Please contact the administrator.";
          return;
        }
        window.location.assign(destination);
      } catch (error) {
        console.error("Portal sign-in failed:", error);
        message.textContent = "Sign-in failed. Check your details or contact the school administrator.";
      }
    });
  </script>
</body>

  <script type="module">
    import { auth, db } from "./firebase-config.js";
    import { onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-auth.js";
    import { doc, getDoc, setDoc, addDoc, collection, serverTimestamp } from "https://www.gstatic.com/firebasejs/13.4.0/firebase-firestore.js";

    onAuthStateChanged(auth, async (user) => {
      if (!user) { window.location.href = "index.html"; return; }
      const uDoc = await getDoc(doc(db, "users", user.uid));
      if (!uDoc.exists() || !["staff", "dos", "headteacher", "director"].includes(uDoc.data().role)) {
        window.location.href = "index.html";
        return;
      }
      const name = uDoc.data().fullName || user.email;
      document.getElementById("staffNavTitle").textContent = `Staff Page: ${name}`;
      document.getElementById("presTeacherName").value = name;
      document.getElementById("lpTeacher").value = name;
      document.getElementById("reqTeacher").value = name;
    });

    document.getElementById("classPresenceForm").addEventListener("submit", async (e) => {
      e.preventDefault();
      const cls = document.getElementById("presClass").value;
      await setDoc(doc(db, "teacher_presence", cls), {
        classLevel: cls,
        teacherName: document.getElementById("presTeacherName").value.trim(),
        subject: document.getElementById("presSubject").value.trim(),
        status: document.getElementById("presStatus").value,
        updatedAt: serverTimestamp()
      });
      document.getElementById("presMsg").textContent = `Presence updated for ${cls} on DOS board!`;
    });

    document.getElementById("lessonPlanForm").addEventListener("submit", async (e) => {
      e.preventDefault();

        await updateDoc(financeSummaryRef, {
          totalBalance: increment(Number(amount))
        });

        // 3. Update Pupil's Fee Balance
        const pupilRef = doc(db, "pupils", pupilId);
        const pupilSnap = await getDoc(pupilRef);
        if (pupilSnap.exists()) {
          const currentBalance = Number(pupilSnap.data().feeBalance || 0);
